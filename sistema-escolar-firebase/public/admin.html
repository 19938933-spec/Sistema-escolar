document.getElementById('btnGenerarPDF').addEventListener('click', async () => { const grado = document.getElementById('selectGradoPDF').value; const periodo = parseInt(document.getElementById('selectPeriodoPDF').value); if (!grado) { mostrarToast('Selecciona un grado', 'error'); return; } mostrarToast('Generando PDF...', 'info'); try { const { jsPDF } = window.jspdf; const pdfDoc = new jsPDF('l', 'mm', 'a4'); // 1. Cargar logo let logoImg = null; try { const logoResp = await fetch('img/fondos/logo.png'); if (logoResp.ok) { const logoBlob = await logoResp.blob(); logoImg = await new Promise((resolve) => { const reader = new FileReader(); reader.onloadend = () => resolve(reader.result); reader.readAsDataURL(logoBlob); }); } } catch (e) { console.log('Logo no disponible'); } // 2. Obtener alumnos del grado const qAlumnos = query(collection(db, 'alumnos'), where('grado', '==', grado)); const snapAlumnos = await getDocs(qAlumnos); const alumnos = []; snapAlumnos.forEach(d => alumnos.push({ id: d.id, ...d.data() })); alumnos.sort((a, b) => a.nombre.localeCompare(b.nombre)); if (alumnos.length === 0) { mostrarToast('No hay alumnos en este grado', 'error'); return; } // 3. Obtener calificaciones Y calcular promedios por materia const qNotas = query(collection(db, 'notas'), where('grado', '==', grado), where('periodo', '==', periodo)); const snapNotas = await getDocs(qNotas); // Estructura: notasPorAlumno[alumnoId][materia] = [nota1, nota2, ...] const notasPorAlumno = {}; const materiasSet = new Set(); snapNotas.forEach(d => { const data = d.data(); if (!notasPorAlumno[data.alumnoId]) notasPorAlumno[data.alumnoId] = {}; if (!notasPorAlumno[data.alumnoId][data.materia]) notasPorAlumno[data.alumnoId][data.materia] = []; notasPorAlumno[data.alumnoId][data.materia].push(parseFloat(data.nota)); materiasSet.add(data.materia); }); const materias = Array.from(materiasSet).sort(); console.log('Materias en PDF:', materias); if (materias.length === 0) { mostrarToast('No hay calificaciones registradas para este período', 'error'); return; } // 4. Función para formatear nombre de materia function formatearNombreMateria(materia) { if (materia.startsWith('modulo_')) { const num = materia.replace('modulo_', ''); return `Módulo ${num}`; } return materia; } // 5. Encabezado del PDF if (logoImg) pdfDoc.addImage(logoImg, 'PNG', 14, 10, 20, 20); pdfDoc.setFontSize(16); pdfDoc.setTextColor(40, 40, 40); pdfDoc.text("Complejo Educativo Thomas Jefferson", logoImg ? 38 : 14, 18); pdfDoc.setFontSize(12); pdfDoc.text("Sistema de Gestión Académica", logoImg ? 38 : 14, 26); pdfDoc.setFontSize(14); pdfDoc.text(`Reporte de Calificaciones - ${grado}`, 14, 42); pdfDoc.setFontSize(10); pdfDoc.setTextColor(100, 100, 100); pdfDoc.text(`Período ${periodo} | Fecha: ${new Date().toLocaleDateString('es-SV')}`, 14, 48); // 6. Construir tabla con promedios finales por materia const headers = [['#', 'Alumno', ...materias.map(m => formatearNombreMateria(m)), 'Prom. Gral.', 'Estado']]; const body = alumnos.map((alumno, idx) => { const row = [idx + 1, alumno.nombre]; let sumaPromedios = 0, countMaterias = 0; materias.forEach(m => { const notasMateria = notasPorAlumno[alumno.id]?.[m] || []; if (notasMateria.length > 0) { const sumaNotas = notasMateria.reduce((sum, nota) => sum + nota, 0); const promedioMateria = sumaNotas / notasMateria.length; row.push(promedioMateria.toFixed(1)); sumaPromedios += promedioMateria; countMaterias++; } else { row.push('-'); } }); const promedioGeneral = countMaterias > 0 ? (sumaPromedios / countMaterias) : 0; row.push(promedioGeneral > 0 ? promedioGeneral.toFixed(1) : '-'); row.push(countMaterias > 0 ? (promedioGeneral >= 7 ? 'Aprobado' : 'Reprobado') : 'Sin notas'); return row; }); // 7. Generar tabla pdfDoc.autoTable({ head: headers, body: body, startY: 55, theme: 'grid', headStyles: { fillColor: [99, 102, 241], textColor: 255, fontStyle: 'bold', fontSize: 9 }, bodyStyles: { fontSize: 9, textColor: 50 }, alternateRowStyles: { fillColor: [240, 240, 245] }, columnStyles: { 0: { cellWidth: 10 }, 1: { cellWidth: 50 } }, didParseCell: function(data) { if (data.section === 'body') { const val = parseFloat(data.cell.raw); if (!isNaN(val) && data.column.index > 1 && data.column.index < data.table.columns.length - 2) { data.cell.styles.textColor = val >= 7.0 ? [16, 185, 129] : [239, 68, 68]; data.cell.styles.fontStyle = 'bold'; } if (data.column.index === data.table.columns.length - 1) { if (data.cell.raw === 'Aprobado') data.cell.styles.textColor = [16, 185, 129]; else if (data.cell.raw === 'Reprobado') data.cell.styles.textColor = [239, 68, 68]; } } } }); // 8. Pie de página const pageCount = pdfDoc.internal.getNumberOfPages(); for (let i = 1; i <= pageCount; i++) { pdfDoc.setPage(i); pdfDoc.setFontSize(8); pdfDoc.setTextColor(150, 150, 150); pdfDoc.text(`Página ${i} de ${pageCount} - Sistema Escolar Thomas Jefferson`, pdfDoc.internal.pageSize.getWidth() - 60, pdfDoc.internal.pageSize.getHeight() - 10); } pdfDoc.save(`Notas_${grado.replace(/[^a-zA-Z0-9]/g, '_')}_P${periodo}.pdf`); mostrarToast('PDF generado correctamente', 'success'); } catch (error) { console.error('Error:', error); mostrarToast('Error al generar PDF: ' + error.message, 'error'); } });
